cmake_minimum_required(VERSION 3.11)
project(reader_instance_purge LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED CONNEXTDDS_DIR AND DEFINED ENV{NDDSHOME})
    set(CONNEXTDDS_DIR "$ENV{NDDSHOME}")
endif()
if(NOT CONNEXTDDS_DIR)
    message(FATAL_ERROR "Set NDDSHOME or CONNEXTDDS_DIR.")
endif()
list(APPEND CMAKE_MODULE_PATH "${CONNEXTDDS_DIR}/resource/cmake")
find_package(RTIConnextDDS "7.7.0" REQUIRED COMPONENTS core)
add_executable(reader_instance_purge reader_instance_purge.cxx)
target_compile_options(reader_instance_purge PRIVATE -Wall -Wextra -Werror)
target_link_libraries(reader_instance_purge RTIConnextDDS::cpp2_api)
