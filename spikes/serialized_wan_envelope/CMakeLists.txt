cmake_minimum_required(VERSION 3.11)
project(serialized_wan_envelope LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "$ENV{NDDSHOME}/resource/cmake")
find_package(RTIConnextDDS "7.7.0" REQUIRED COMPONENTS core)
foreach(role application gateway)
    add_executable(${role} ${role}.cxx)
    target_compile_options(${role} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(${role} RTIConnextDDS::cpp2_api)
endforeach()
