cmake_minimum_required(VERSION 3.11)
project(union_auto_qos LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "$ENV{NDDSHOME}/resource/cmake")
find_package(RTIConnextDDS "7.7.0" REQUIRED COMPONENTS core routing_service)
add_library(union_auto_qos SHARED Processor.cxx)
target_compile_options(union_auto_qos PRIVATE -Wall -Wextra -Werror)
target_link_libraries(union_auto_qos PRIVATE RTIConnextDDS::routing_service_cpp2)
