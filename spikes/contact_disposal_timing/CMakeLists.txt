cmake_minimum_required(VERSION 3.11)
project(contact_disposal_timing LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
list(APPEND CMAKE_MODULE_PATH "$ENV{NDDSHOME}/resource/cmake")
find_package(RTIConnextDDS "7.7.0" REQUIRED COMPONENTS core)
add_executable(contact_disposal_timing timing.cxx)
target_compile_options(contact_disposal_timing PRIVATE -Wall -Wextra -Werror)
target_link_libraries(contact_disposal_timing RTIConnextDDS::cpp2_api)
add_library(contact_cache_monitor SHARED cache_monitor.c)
target_compile_options(contact_cache_monitor PRIVATE -Wall -Wextra -Werror)
target_link_libraries(contact_cache_monitor RTIConnextDDS::c_api pthread)
add_executable(ddil_probe ddil_probe.cxx)
target_compile_options(ddil_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ddil_probe RTIConnextDDS::cpp2_api)
